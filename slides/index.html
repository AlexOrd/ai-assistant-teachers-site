function doPost(e) {
  // 1. секретний ключ в URL
  if (e.parameter.k !== prop_('WEBHOOK_KEY'))
    return ok_();
  const update = JSON.parse(e.postData.contents);
  // 2. повтори пропускаємо
  if (seen_(update.update_id)) return ok_();
  handleMessage_(update.message);
  return ok_();
}
